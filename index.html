function doPost(e){
 var d=JSON.parse(e.postData.contents);
 var ss=SpreadsheetApp.getActiveSpreadsheet();
 var sh=ss.getSheetByName(d.cat)||ss.insertSheet(d.cat);
 if(sh.getLastRow()===0) sh.appendRow(['시간','내용','날짜태그','긴급']);
 sh.appendRow([d.time,d.text,d.dates,d.urgent]);
 return ContentService.createTextOutput('ok');
}
